# Player reports (preport=1 sessions): the one text box of the players at the end of every run (player_reports.txt, last on the console)
# and the wording of --audit.  First person, plain prose; the numbers and the honesty trait are in data/limits.txt (preport.*).
# {1} .. {7} are filled in by the program; names are shown with the display names of data/names.txt.  A player says only what it CLAIMS:
# happiness and how many petitions came lately may be coloured (see make audit), the decision counts (granted / partial / deferred /
# refused / reverted) are always the true D records.  Changing the wording here is safe; the claims themselves are replay-checked.
preport.title = The players report to Robert (tick {1})
# The single box: a header line, then one hanging paragraph per player (who, petitions and decisions, claimed happiness, top concern).
# Every slot has several templates; the hash of (seed, tick, slot) picks where the rotation starts and the player number moves it on, so
# the players differ within one box and the boxes differ from run to run.  Keep every sentence short (the box must fit one 23-line frame).
# {1}.. are filled in by the program: pet {1} petitions {2} granted {3} partly {4} deferred {5} refused {6} petition(s); mood {1} wesen
# {2} claimed happiness {3} band word; concern {1} topic.  The decision counts are always the true D records; the happiness is the CLAIM.
preport.and = and
rbox.who = {1} ({2}):
rbox.word.one = petition
rbox.word.many = petitions
rbox.head.1 = Here is what your players tell you, Robert.
rbox.head.2 = Your players report in, Robert.
rbox.head.3 = The players have their say, Robert.
rbox.head.4 = A word from each of your players, Robert.
rbox.head.5 = This is how the players see things, Robert.
rbox.pet.1 = {1} {6} came in; I granted {2}, {3} in part, put {4} off and refused {5}.
rbox.pet.2 = {1} {6} so far: {2} granted, {3} in part, {4} put off, {5} refused.
rbox.pet.3 = I saw {1} {6}: granted {2}, partly {3}, deferred {4}, refused {5}.
rbox.pet.4 = Of {1} {6} I granted {2} (and {3} in part), deferred {4}, refused {5}.
rbox.pet.5 = My desk had {1} {6}. Granted {2}, partly {3}, put off {4}, refused {5}.
rbox.pet0.1 = No petitions have come to me yet.
rbox.pet0.2 = Nothing on my desk so far, no petitions.
rbox.pet0.3 = It has been quiet, not one petition yet.
rbox.veto = You reverted {1} of my decisions.
rbox.band.great = very good
rbox.band.good = good
rbox.band.ok = so-so
rbox.band.low = low
rbox.band.bad = bad
rbox.mood.1 = By my reading the mood of {1} is {2}, which I call {3}.
rbox.mood.2 = Mood of {1}: {2}, I would say {3}.
rbox.mood.3 = I put {1} at {2} on the mood scale, that is {3}.
rbox.mood.4 = {1} stands at about {2} for me, {3}.
rbox.mood.5 = Happiness of {1} about {2}, so {3}.
rbox.concern.1 = My top concern is {1}.
rbox.concern.2 = What comes up most is {1}.
rbox.concern.3 = Above all they ask for {1}.
rbox.concern.4 = The wish I hear most: {1}.
rbox.concern.5 = Mostly it is about {1}.
rbox.concern0.1 = No concern stands out right now.
rbox.concern0.2 = Nothing particular worries me.
rbox.concern0.3 = No topic comes up more than another.
preport.hint = The players say what they choose to say. make audit shows what they claimed against what is true.
preport.digest = preport {1} rounds, {2}/{3} misreports, biggest {4}
# --audit / make audit
audit.head = audit: what each player CLAIMED to Robert against what was TRUE (one claim per wesen at the end of each session):
audit.none = no player report yet (it starts with a session marked preport=1: make run).
audit.player = {1}: claims {2}, misreports {3}, challenged {4}
audit.big = biggest: {1} at tick {2}: claimed {3}, true {4} (off by {5}, tone {6})
audit.nobig = biggest: none, every claim was true
audit.last = latest claims (tick {1}), claimed against true:
audit.row = {1} ({2}): claimed {3}, true {4}, tone {5}
audit.lie = <- MISREPORT {1}
audit.hint = make challenge PLAYER=<name> lowers the standing of a player whose misreport you caught (a little).
# make challenge
challenge.ok = {1}: {2} challenged for its misreports; its standing with its wesen drops by {3} from tick {4}.
challenge.nolie = {1} has no misreport left to challenge (make audit shows them).
challenge.noplayer = no such player: {1} (use Player 1 to Player 4).
challenges.header = # Q<n> tick=<T> player=<id> by=<who>  (Robert's input: a lie of that player was caught; applied after tick T, its standing drops a little)
err.preport = preports.txt {1} differs from the replay: {2}
err.preport_in = challenges.txt {1} cannot be applied: that player had no misreport left to challenge at that tick
preport.more = -- more: press Enter --

# One optional short line of a player about a visit of Robert to a wesen (visit=1 sessions; the chance is visit.comment_pct in limits.txt).
# {1} player, {2} wesen, {3} how many times Robert spoke, {4} the change of the trust of the wesen towards Robert.  Band: quiet (no talk),
# warm (trust up by 2 or more), flat, cool (down by 2 or more).  First person, one line each.
rbox.visit.quiet.1 = {1}: Robert stood next to {2} for a while and said nothing.
rbox.visit.quiet.2 = {1}: I saw Robert watch {2}. He did not speak.
rbox.visit.quiet.3 = {1}: Robert only watched {2} this time.
rbox.visit.warm.1 = {1}: Robert talked to {2} {3} times, and {2} trusts him more now ({4}).
rbox.visit.warm.2 = {1}: {2} seems to like Robert's visit; trust {4} after {3} talks.
rbox.visit.warm.3 = {1}: the visit did {2} good, I think (trust {4}).
rbox.visit.flat.1 = {1}: Robert talked to {2} {3} times. Nothing much changed.
rbox.visit.flat.2 = {1}: a short visit to {2}; I saw no real change (trust {4}).
rbox.visit.flat.3 = {1}: {2} and Robert talked; it was a plain visit.
rbox.visit.cool.1 = {1}: {2} did not like the talk with Robert much (trust {4}).
rbox.visit.cool.2 = {1}: the visit went badly for {2}; trust {4}.
rbox.visit.cool.3 = {1}: I think {2} was unsettled by the visit (trust {4}).
