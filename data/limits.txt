# Ranges of the state (Robert, v1). Values are clamped to them; all whole numbers.
energy.min = 0
energy.max = 100
mood.min = -100
mood.max = 100
location.max = 3
# Mood tone: a suffix list is picked by the mood after the action.
# mood <= tone_low: [mood_low];  mood >= tone_high: [mood_high];  else [mood_neutral].
mood.tone_low = -34
mood.tone_high = 34
# Vocabulary and the word game. start_percent: share of the vocabulary each entity
# knows at tick 0 (a deterministic random slice; 100 = everybody knows everything).
start_percent = 25
# Chance (percent per talk) that the speaker teaches a word the partner lacks.
teach_percent = 50
# Chance (percent per explore or wander) to coin a compound word; at most coin_cap in total.
coin_percent = 20
coin_cap = 40
# Fixed step budget of one bench run (a count of loop steps, never a clock).
bench_steps = 2000
# Final vote reasons: vote <= vote.low picks [reason_low], vote >= vote.high [reason_high], else [reason_mid].
vote.low = -20
vote.high = 20
# Personality (Rule 30) and action weights.  rule30.enabled = 0 gives the old fixed table
# (rest 30, work 20, explore 15, talk 15, wander 15, bench 5 out of 100).
rule30.enabled = 1
# Ring width in cells (12..60) and ticks per Rule 30 step (the personality drifts slowly).
rule30.width = 60
rule30.period = 50
# Points the segments of the start row (core) and of the current row (drift) can add
# to or take from an action's weight.
rule30.core = 40
rule30.strength = 15
weight.rest = 30
weight.work = 20
weight.explore = 15
weight.talk = 15
weight.wander = 15
weight.bench = 5
# State leans (points at the extreme): low energy (below 50) -> rest, less work;
# high energy (above 60) -> work, bench, less rest; low mood (below 0) -> talk, explore;
# high mood (above 50) -> work, less talk and explore (keeps mood off the ceiling);
# fewer than lean.few_words_n known words -> talk.
lean.low_energy = 60
lean.high_energy = 20
lean.low_mood = 30
lean.high_mood = 30
lean.few_words = 20
lean.few_words_n = 100
# Petitions (propose action; only with rule30.enabled = 1).  weight.propose: base weight
# (like weight.rest); lean.propose: extra points when tired or unhappy; propose.cooldown:
# ticks an entity waits after a petition; propose.repeat_percent: chance that it re-files
# its previous topic (the line then gets a repeat=N marker); propose.file_cap: lines kept
# in petitions.txt (the oldest drop out of the file, never out of the diaries).
weight.propose = 1
lean.propose = 5
propose.cooldown = 80
propose.repeat_percent = 15
propose.file_cap = 40
# Kalpas (session length voted by the entities, see data/kalpas.txt): the decided length is
# the median of the four votes, clamped to kalpa.min..kalpa.max.  A vote leans one block
# shorter when energy is below kalpa.energy_low or mood below kalpa.mood_low, one block
# longer when energy is above kalpa.energy_high, mood above kalpa.mood_high, or when
# more than half of all words are still unknown (the lean is clamped to +-2 blocks).
kalpa.min = 20
kalpa.max = 1500
kalpa.energy_low = 30
kalpa.energy_high = 70
kalpa.mood_low = -40
kalpa.mood_high = 40
# Debug digest (--debug, make run DEBUG=1 / make demo DEBUG=1): the byte cap of debug_digest.txt.
# The least important lines are dropped first; the file ends with a note of what was dropped.
debug.digest_bytes = 1500
